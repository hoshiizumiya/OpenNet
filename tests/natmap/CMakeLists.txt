cmake_minimum_required(VERSION 3.20)
project(OpenNetNatMapTests LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NATMAP_SANITIZERS "Instrument the packet parser with ASan and UBSan" OFF)
add_executable(stun_protocol_tests StunProtocolTests.cpp)
target_include_directories(stun_protocol_tests PRIVATE ../../OpenNet)
if(MSVC)
  target_compile_options(stun_protocol_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(stun_protocol_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(NATMAP_SANITIZERS)
    target_compile_options(stun_protocol_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(stun_protocol_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME stun_protocol COMMAND stun_protocol_tests)
if(WIN32)
  add_executable(udp_mapping_tests UdpMappingTests.cpp
    ../../OpenNet/Core/NatMap/UdpMappingService.cpp
    ../../OpenNet/Core/NatMap/MappingFirewall.cpp)
  target_include_directories(udp_mapping_tests PRIVATE ../../OpenNet)
  target_compile_definitions(udp_mapping_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(udp_mapping_tests PRIVATE /W4 /WX /permissive-)
  target_link_libraries(udp_mapping_tests PRIVATE ws2_32 bcrypt ole32 oleaut32)
  add_test(NAME udp_mapping COMMAND udp_mapping_tests)
  set_tests_properties(udp_mapping PROPERTIES TIMEOUT 30)

  add_executable(tcp_port_reuse_tests TcpPortReuseTests.cpp)
  target_include_directories(tcp_port_reuse_tests PRIVATE ../../OpenNet)
  target_compile_definitions(tcp_port_reuse_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(tcp_port_reuse_tests PRIVATE /W4 /WX /permissive-)
  target_link_libraries(tcp_port_reuse_tests PRIVATE ws2_32)
  add_test(NAME tcp_port_reuse COMMAND tcp_port_reuse_tests)
  set_tests_properties(tcp_port_reuse PROPERTIES TIMEOUT 20)

  add_executable(tcp_mapping_tests TcpMappingTests.cpp
    ../../OpenNet/Core/NatMap/TcpMappingService.cpp)
  target_include_directories(tcp_mapping_tests PRIVATE ../../OpenNet)
  target_compile_definitions(tcp_mapping_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(tcp_mapping_tests PRIVATE /W4 /WX /permissive-)
  target_link_libraries(tcp_mapping_tests PRIVATE ws2_32 bcrypt)
  add_test(NAME tcp_mapping COMMAND tcp_mapping_tests)
  set_tests_properties(tcp_mapping PROPERTIES TIMEOUT 30)
endif()
