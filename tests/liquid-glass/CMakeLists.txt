cmake_minimum_required(VERSION 3.20)
project(OpenNetLiquidGlassTests LANGUAGES CXX)
include(CTest)
add_executable(liquid_glass_parameter_tests ParameterStateTests.cpp)
target_include_directories(liquid_glass_parameter_tests PRIVATE ../../OpenNet)
target_compile_features(liquid_glass_parameter_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(liquid_glass_parameter_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(liquid_glass_parameter_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME liquid_glass_parameters COMMAND liquid_glass_parameter_tests)
